def counts_to_dataframe(stats_dict, sample_id, aligner_name):
    """
    Convert the aggregated stats dict into a tidy DataFrame.
    stats_dict example:
      {
        "TP_insertion": {1:5, 2:10, 3:1, ...},
        "TP_deletion":  {...},
        "FP_insertion": {...},
        "FP_deletion":  {...},
        "FN_insertion": {...},
        "FN_deletion":  {...},
      }

    Returns a DataFrame with columns:
      [sample_id, aligner_name, classification, variant_type, length, count]
    """
    rows = []
    for classification_varType, length_counts in stats_dict.items():
        # classification_varType => e.g. "TP_insertion"
        classification, var_type = classification_varType.split("_")
        for length, count in length_counts.items():
            rows.append({
                "sample_id": sample_id,
                "aligner_name": aligner_name,
                "classification": classification,   # TP, FP, or FN
                "variant_type": var_type,          # insertion or deletion
                "length": length,
                "count": count
            })
    return pd.DataFrame(rows)

def parse_isec_vcfs_indel_length_counts(isec_dir):
    """
    Parse bcftools isec output (0000, 0001, 0002) and aggregate counts of
    insertion/deletion lengths for TP, FP, FN.
    Returns a dictionary of dictionaries, e.g.:
      {
        "TP_insertion": { length1: count1, length2: count2, ... },
        "TP_deletion":  { ... },
        "FP_insertion": { ... },
        "FP_deletion":  { ... },
        "FN_insertion": { ... },
        "FN_deletion":  { ... }
      }
    """
    stats = {
        "TP_insertion": defaultdict(int),
        "TP_deletion": defaultdict(int),
        "FP_insertion": defaultdict(int),
        "FP_deletion": defaultdict(int),
        "FN_insertion": defaultdict(int),
        "FN_deletion": defaultdict(int),
    }

    vcf_0000 = os.path.join(isec_dir, "0000.vcf")  # Unique to first => FP
    vcf_0001 = os.path.join(isec_dir, "0001.vcf")  # Unique to second => FN
    vcf_0002 = os.path.join(isec_dir, "0002.vcf")  # Common => TP

    def classify_variant(ref, alt):
        if len(alt) > len(ref):
            return "insertion"
        elif len(ref) > len(alt):
            return "deletion"
        else:
            return "other"

    def parse_vcf(vcf_path, classification):
        if not os.path.exists(vcf_path):
            return
        with open(vcf_path, "r") as f:
            for line in f:
                if line.startswith("#"):
                    continue
                fields = line.strip().split("\t")
                ref = fields[3]
                alt = fields[4]
                var_type = classify_variant(ref, alt)
                if var_type in ("insertion", "deletion"):
                    indel_len = abs(len(alt) - len(ref))
                    key = f"{classification}_{var_type}"  # e.g. "FP_insertion"
                    stats[key][indel_len] += 1

    parse_vcf(vcf_0000, "FP")
    parse_vcf(vcf_0001, "FN")
    parse_vcf(vcf_0002, "TP")

    return stats

def parse_isec_vcfs(isec_dir, first_vcf_label="aligner", second_vcf_label="ground_truth"):

      - TP_insertion, TP_deletion
      - FP_insertion, FP_deletion
      - FN_insertion, FN_deletion
    The assumption is:
      0000.vcf => present only in first VCF  (FP if the first VCF is the 'aligner')
      0001.vcf => present only in second VCF (FN if second VCF is 'ground_truth')
      0002.vcf => present in both           (TP)
    """

    vcf_0000 = os.path.join(isec_dir, "0000.vcf")
    vcf_0001 = os.path.join(isec_dir, "0001.vcf")

    # Initialize counters
    stats = {
        "TP_insertion": 0,
        "TP_deletion": 0,
        "FP_insertion": 0,
        "FP_deletion": 0,
        "FN_insertion": 0,
        "FN_deletion": 0,
    }

    # Helper function to classify a single REF/ALT
    def classify_variant(ref, alt):
        if len(alt) > len(ref):
            return "insertion"
        elif len(ref) > len(alt):
            return "deletion"
        else:
            return "other"  # e.g., SNP or complex if len same but different bases

    # If file not present, skip. Could happen if no variants are in that category.
    if os.path.exists(vcf_0000):
        with open(vcf_0000) as f:
            for line in f:
                if line.startswith("#"):
                    continue
                fields = line.strip().split("\t")
                ref = fields[3]
                alt = fields[4]
                variant_type = classify_variant(ref, alt)
                if variant_type == "insertion":
                    stats["FP_insertion"] += 1
                elif variant_type == "deletion":
                    stats["FP_deletion"] += 1

    if os.path.exists(vcf_0001):
        with open(vcf_0001) as f:
            for line in f:
                if line.startswith("#"):
                    continue
                fields = line.strip().split("\t")
                ref = fields[3]
                alt = fields[4]
                variant_type = classify_variant(ref, alt)
                if variant_type == "insertion":
                    stats["FN_insertion"] += 1
                elif variant_type == "deletion":
                    stats["FN_deletion"] += 1

    if os.path.exists(vcf_0002):
        with open(vcf_0002) as f:
            for line in f:
                if line.startswith("#"):
                    continue
                fields = line.strip().split("\t")
                ref = fields[3]
                alt = fields[4]
                variant_type = classify_variant(ref, alt)
                if variant_type == "insertion":
                    stats["TP_insertion"] += 1
                elif variant_type == "deletion":
                    stats["TP_deletion"] += 1

    return stats

def plot_indel_length_grid(df, sample_id, output_dir, log_scale=False):
    """
    Create a 2x3 grid of plots:
      Rows = variant_type: [insertion, deletion]
      Columns = classification: [TP, FP, FN]

    Each subplot has lines for each aligner. X-axis: Indel length, Y-axis: count.
    All x axes share the same ticks (step=1), and we draw vertical grid lines.
    """

    # Desired order of aligners
    aligner_order = ["DeepSAP", "DRAGEN", "novoSplice", "STAR", "HISAT2", "Subjunc"]

    # Rows, columns definitions
    variant_types = ["insertion", "deletion"]     # row 0 => insertion, row 1 => deletion
    classifications = ["TP", "FP", "FN"]          # col 0 => TP, col 1 => FP, col 2 => FN

    fig, axes = plt.subplots(nrows=2, ncols=3, sharex='col', sharey=True, figsize=(18, 8))

    for i, var_type in enumerate(variant_types):
        for j, classif in enumerate(classifications):
            ax = axes[i, j]

            # Filter the DataFrame
            sub = df[
                (df["sample_id"] == sample_id) &
                (df["classification"] == classif) &
                (df["variant_type"] == var_type)
            ]
            if sub.empty:
                ax.set_title(f"{classif} - {var_type}\n(No data)")
                ax.set_xlabel("Indel Length")
                ax.set_ylabel("Count")
                continue

            # Group by (aligner_name, length), sum counts
            grouped = sub.groupby(["aligner_name", "length"], as_index=False)["count"].sum()

            # Pivot so that index=length, columns=aligner_name, values=count
            pivoted = grouped.pivot(index="length", columns="aligner_name", values="count").fillna(0)

            # Plot lines in desired aligner order
            for aligner in aligner_order:
                if aligner in pivoted.columns:
                    ax.plot(pivoted.index, pivoted[aligner], label=aligner)

            # Title
            ax.set_title(f"{classif} - {var_type}")

            # Set x/y labels only for bottom/left edges
            if i == 1:
                ax.set_xlabel("Indel Length")
            if j == 0:
                ax.set_ylabel("Count")

            # 1) Ensure x ticks at every integer (step=1).
            #   If pivoted.index has min=0, max=20, for example, we’ll see ticks at 0..20.
            ax.xaxis.set_major_locator(mticker.MultipleLocator(2))

            # 2) Draw vertical grid lines at each tick (which='major')
            ax.grid(True, axis='x', which='major', linestyle='--', alpha=0.7)

            if log_scale:
                ax.set_yscale("log")

    
    for row in range(2):
        for col in range(3):
            axes[row, col].tick_params(labelbottom=True)

    # Let's do a suptitle as well
    fig.suptitle(f"Indel length trends in {sample_id}", y=1.02, fontsize=14)

    # (2) Collect legend handles/labels (from any subplot that has them):
    handles, labels = axes[0,0].get_legend_handles_labels()

    # (3) Call tight_layout so subplots are nicely spaced
    plt.tight_layout()

    # (4) Expand the bottom margin a bit so the legend fits
    #     The bigger the number, the more space below the subplots.
    fig.subplots_adjust(bottom=0.10)

    # (5) Add the legend inside the figure, at the lower center, no bounding box anchor needed
    fig.legend(handles, labels, loc="lower center", ncol=len(labels))

    plt.show()

    fig.savefig(output_dir + f"/indels_events_length_{sample_id}.png", dpi=300, bbox_inches='tight')

def reverse_complement(dna: str) -> str:
    """
    Returns the reverse complement of a given DNA sequence.
    
    Parameters:
        dna (str): Input DNA sequence (A, T, C, G)
    
    Returns:
        str: Reverse complement of the DNA sequence
    """
    complement = {
        'A': 'T', 'T': 'A', 'C': 'G', 'G': 'C', 'N': 'N',
        'a': 't', 't': 'a', 'c': 'g', 'g': 'c'  # Handles lowercase input
    }
    
    # Convert to reverse complement
    return "".join(complement[base] for base in reversed(dna))

def generate_paired_sam(input_file, output_sam_file, sample_id):
    
    if "malaria" in sample_id:
        header    = header_malaria
    else:
        header    = header_human

    with open(output_sam_file, "w") as sam_file:
        # Write SAM header
        sam_file.write(header)
        # Read ground truth data
        with open(input_file, "r") as f:
            pairs = {}
            for line in f:
                fields = line.strip().split("\t")
                seq_id = fields[0][:-1]  # Remove trailing 'a' or 'b'
                pair_type = fields[0][-1]  # 'a' or 'b'
                chromosome = fields[1]
                pos = int(fields[2])
                end = int(fields[3])
                cigar = fields[4]
                seq = fields[7]
                strand = fields[6]

                # Compute FLAG
                if pair_type == 'a':
                    if strand == '+':
                        flag = 99 
                    else:
                        flag = 83
                        seq = reverse_complement(seq)

                elif pair_type == 'b':
                    if strand == '+':
                        flag = 163 
                    else:
                        flag = 147
                        seq = reverse_complement(seq)

                read = {
                    "seq_id": seq_id,
                    "chromosome": chromosome,
                    "pos": pos,
                    "cigar": cigar,
                    "seq": seq,
                    "flag": flag,
                    "qual": ":" * len(seq)
                }

                if seq_id not in pairs:
                    pairs[seq_id] = {}
                pairs[seq_id][pair_type] = read

            # Write paired-end reads
            for seq_id, reads in pairs.items():
                if "a" in reads and "b" in reads:
                    read_a = reads["a"]
                    read_b = reads["b"]

                    # Calculate TLEN
                    tlen = abs(read_b["pos"] + len(read_b["seq"]) - read_a["pos"])
                    tlen = tlen if read_a["pos"] < read_b["pos"] else -tlen

                    # Write both reads
                    sam_file.write(
                        f"{read_a['seq_id']}a\t{read_a['flag']}\t{read_a['chromosome']}\t{read_a['pos']}\t255\t{read_a['cigar']}\t=\t{read_b['pos']}\t{tlen}\t{read_a['seq']}\t{read_a['qual']}\n"
                    )
                    sam_file.write(
                        f"{read_b['seq_id']}b\t{read_b['flag']}\t{read_b['chromosome']}\t{read_b['pos']}\t255\t{read_b['cigar']}\t=\t{read_a['pos']}\t{-tlen}\t{read_b['seq']}\t{read_a['qual']}\n"
                    )
                else:
                    print(f"Unpaired read found for {seq_id}")

def convert_sam_to_bam(input_sam_file, output_bam_file):
    samfile = pysam.AlignmentFile(input_sam_file, "r")
    bamfile = pysam.AlignmentFile(output_bam_file, "wb", header=samfile.header)
    
    for read in samfile:
        bamfile.write(read)
    
    samfile.close()
    bamfile.close()

def run_varscan_pipeline(aligner_name, bam_file, sample_id, sample_output_folder, reference_fasta_human, reference_fasta_malaria, varscan_args=[]):
    """
    This function runs:
      1) samtools sort
      2) samtools index
      3) samtools mpileup + varscan
    for one (aligner, sample) pair.
    """

    if "malaria" in sample_id:
        reference_fasta = reference_fasta_malaria
    else:
        reference_fasta = reference_fasta_human
    
    sorted_bam = os.path.join(sample_output_folder, f"{aligner_name}_{sample_id}.sorted.bam")
    varscan_out_vcf = os.path.join(sample_output_folder, f"{aligner_name}_{sample_id}.varscan.indels.vcf")
    process_log_out = os.path.join(sample_output_folder, f"{aligner_name}_{sample_id}.log")

    print(f"  === [Thread] Processing {aligner_name} for sample {sample_id} ===")
    print(f"      BAM input: {bam_file}")

    # 1A: Sort the BAM
    subprocess.run([
        "samtools", "sort",
        "-@", "1", 
        "-o", sorted_bam,
        bam_file
    ], check=True)

    # 1B: Index the sorted BAM
    subprocess.run([
        "samtools", "index",
        sorted_bam
    ], check=True)

    # 1C: mpileup -> varscan
    mpileup_cmd = [
        "samtools", "mpileup",
        "-f", reference_fasta,
        sorted_bam
    ]

    varscan_cmd = [
        "varscan", "mpileup2indel",
        "--output-vcf", "1",
    ]

    varscan_cmd += varscan_args

    print(f"      VarScan output VCF: {varscan_out_vcf}")

    with open(process_log_out, "w") as log_out:
        with open(varscan_out_vcf, "w") as vcf_out:
            p1 = subprocess.Popen(mpileup_cmd, stdout=subprocess.PIPE, stderr=log_out)
            p2 = subprocess.Popen(varscan_cmd, stdin=p1.stdout, stdout=vcf_out, stderr=log_out)
            p1.stdout.close()
            p2.communicate()

    return varscan_out_vcf  # If you want to retrieve this later
