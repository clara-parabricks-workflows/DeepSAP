def benchmark_sample_bam(aligner_name: str, bam_path: Path, sample_id: str):

    log_str += f"Finished converting to sam {sam_file}\n"

    # Fix SAM
    with fixed_sam.open("w") as fh:

    log_str += f"Finished fixing sam {fixed_sam}\n"

    # Benchmark
    # with bench_out.open("w") as fh:
    #     subprocess.run(["ruby", compareToTruth, str(cig_path), str(fixed_sam)], stdout=fh, stderr=fh, check=True)

        p1 = subprocess.Popen(
            ["ruby", compareToTruth, str(cig_path), str(fixed_sam)],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE
        )
        p2 = subprocess.Popen(

            stdin=p1.stdout,
            stdout=fout

    log_str += f"Finished benchmarking {bench_out}\n"

        sam_file.unlink()

        log_str += f"⚠️ Failed to delete temp files: {e}\n"
    # print(log_str)
    return log_str

def parse_number(val: str):
    """
    Parse numeric string as int or float, handles 'NaN'
    """
    val = val.strip()
    if val == 'NaN':
        return float('nan')
    return float(val) if '.' in val or 'e' in val.lower() else int(val)

def extract_record(text_block, patterns):
    record = {}
    for field, pat in patterns.items():
        match = re.search(pat, text_block, re.MULTILINE | re.DOTALL)
        if match:
            if len(match.groups()) == 1:
                val = match.group(1)
                record[field] = float(val) if 'NaN' in val or '.' in val else int(val)
            else:
                for i, g in enumerate(match.groups()):
                    record[f"{field}_{i}"] = float(g) if 'NaN' in g or '.' in g else int(g)

    if record:
        values_to_check = {
            k: v for k, v in record.items()
            if not k.startswith("junctions_sides_pct") and isinstance(v, (int, float))
        }
        for v in values_to_check.values():
            if isinstance(v, float) and math.isnan(v):
                continue
            if v < 0:
                return None
        return record
    return None

        n_subsets (int): Number of bootstrap replicates to generate
        sample_size (int): Number of samples per bootstrap replicate

    Returns:
        pd.DataFrame: One row per bootstrap replicate with aggregated metrics
    """
    # Load input data
    df = pd.read_csv(csv_gz_path, compression="gzip")

    # Separate columns into percentage-like and count-like
    percent_cols = [col for col in df.columns if df[col].dtype in [float, np.float64]]
    count_cols   = [col for col in df.columns if df[col].dtype in [int, np.int64]]

    boot_data = []

    for i in range(n_subsets):
        sample = df.sample(n=sample_size, replace=True, random_state=777 + i)
        row = {}

        for col in percent_cols:
            row[col] = sample[col].mean()

        for col in count_cols:
            row[col] = sample[col].sum()

        row["supplest"] = i
        row["dataset"] = dataset
        row["aligner"] = aligner

        boot_data.append(row)

    pd.DataFrame(boot_data).to_csv(csv_out_path,index=False, compression="gzip")

    return f"✅ Bootstrapped sample_size {sample_size}, n_subsets {n_subsets} → {csv_out_path}"


def compute_bootstrap_row(df: pd.DataFrame,
                          sample_size: int,
                          replicate_id: int,
                          dataset: str,
                          n_subsets:int,
                          aligner: str) -> dict:
    # print(f"\t▶ Started computing replicate_id: {replicate_id} for {dataset}/{aligner}")
    rng = np.random.default_rng(seed=777 + replicate_id)
    idx = rng.choice(df.index, size=sample_size, replace=True)

    percent_cols = df.select_dtypes(include=["float"]).columns
    count_cols   = df.select_dtypes(include=["int"]).columns

    sample = df.loc[idx]
    row = {col: sample[col].mean() for col in percent_cols}
    row.update({col: sample[col].sum() for col in count_cols})
    row["supplest"] = replicate_id
    row["dataset"] = dataset
    row["aligner"] = aligner

    if (replicate_id + 1) % 10 == 0:
        print(f"\t▶ Finished replicate {replicate_id + 1} / {n_subsets} for {dataset}/{aligner}")


    return row

def run_bootstrap_for_file(csv_gz_path: str,
                           csv_out_path: str,
                           dataset: str,
                           aligner: str,
                           n_subsets: int,
                           sample_size: int,
                           parallel: bool = True,
                           max_workers: int = 8) -> str:
    print(f"▶ Bootstrapping {dataset}/{aligner}")

    df = pd.read_csv(csv_gz_path, compression="gzip", low_memory=False)
    print(f"▶ Finished reading df: {csv_gz_path}")

    tasks = [(df, sample_size, i, dataset, n_subsets, aligner) for i in range(n_subsets)]

    from concurrent.futures import ThreadPoolExecutor

    if parallel:
        with ThreadPoolExecutor(max_workers=max_workers) as executor:
            rows = list(executor.map(lambda args: compute_bootstrap_row(*args), tasks))

    else:
        rows = [compute_bootstrap_row(*args) for args in tasks]

    # Write to .csv.gz
    pd.DataFrame(rows).to_csv(csv_out_path, index=False, compression="gzip")

    return f"✅ Completed {dataset}/{aligner} → {csv_out_path}"

import os
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from pathlib import Path
from typing import List, Literal, Dict

def plot_all_stacked_bootstrap_histograms(
    datasets:  List[str],
    aligners:  List[str],
    base_dir:  str,
    output_pdf: str,
    output_csv:str,
    n_subsets: int,
    sample_size: int,
    # *,
    method:    Literal["counts", "rates"] = "counts",
    focus:     Literal["segment", "total"] = "segment",
    anchoring: Literal["center", "top", None] = None,
    ci_mode:   Literal["width", "fullrange"] = "width",
   
):
    """
    Grouped stacked-bar plot with bootstrap CIs.

    Parameters
    ----------
    method : "counts" | "rates"
        "counts"  → plot absolute base numbers;
        "rates"   → divide each replicate by its own total, plotting proportions.
    focus  : "segment" | "total"
        "segment" → draw CI for each coloured slice;
        "total"   → draw *one* CI for the cumulative bar height, computed
                    per replicate to include covariance between slices.
    anchoring : "center" | "top" | None
        Where to place the error bar.  None = sensible default
        (center for segment CIs, top for total CIs).
    ci_mode : "width" | "fullrange"
        Either ±½-width or full 95 % range.
    """

    if anchoring is None:
        anchoring = "center" if focus == "segment" else "top"

    abbrev: Dict[str, str] = {
        "baruzzo_human_t1r1":  "HT1",
        "baruzzo_human_t2r1":  "HT2",
        "baruzzo_human_t3r1":  "HT3",
        "baruzzo_malaria_t1r1": "MT1",
        "baruzzo_malaria_t2r1": "MT2",
        "baruzzo_malaria_t3r1": "MT3",
    }

    required_cols = [
        "bases_aligned_correct",
        "bases_aligned_ambiguous",
        "bases_aligned_incorrect",
        "bases_unaligned",
    ]
    colors  = ["#019E72", "#E59F00", "#CE3701", "#C0C0C0"]
    labels  = ["Correct", "Ambiguous", "Incorrect", "Unaligned"]

    # ------------------------------------------------------------------ #
    # Step 1: gather bootstrap summaries
    # ------------------------------------------------------------------ #
    plot_data = []           # per aligner-dataset, store means + CI info
    col_tot   = "bases_total"

    for aligner in aligners:
        for dataset in datasets:
            csv_path = (
                Path(base_dir)
                / dataset
                / aligner
                / f"{dataset}_{aligner}.bootstraps.csv.gz"
            )
            if not csv_path.is_file():
                print(f"⚠️ Missing file: {csv_path}")
                continue

            df = pd.read_csv(csv_path, compression="gzip", low_memory=False)

            if not all(c in df.columns for c in required_cols):
                print(f"⚠️ Missing columns in {csv_path.name}")
                continue

            # convert to rates if asked -----------------------------------------
            if method == "rates":
                df[col_tot] = df[required_cols].sum(axis=1)
                df[required_cols] = df[required_cols].div(df[col_tot], axis=0)

            # stats per coloured slice ------------------------------------------
            mean_seg  = df[required_cols].mean()
            lower_seg = df[required_cols].quantile(0.025)
            upper_seg = df[required_cols].quantile(0.975)

            # stats for *total* bar height (captures covariance) ----------------
            df["bar_height"] = df[required_cols].sum(axis=1)
            mean_tot  = df["bar_height"].mean()
            lower_tot = df["bar_height"].quantile(0.025)
            upper_tot = df["bar_height"].quantile(0.975)

            plot_data.append(
                dict(
                    aligner   = aligner,
                    dataset   = dataset,
                    abbrev    = abbrev.get(dataset, dataset),
                    mean_seg  = mean_seg,
                    lower_seg = lower_seg,
                    upper_seg = upper_seg,
                    mean_tot  = mean_tot,
                    lower_tot = lower_tot,
                    upper_tot = upper_tot,
                )
            )

    flat_data_for_csv = []
    for item in plot_data:
        # Start with the simple key-value pairs
        base_info = {
            'aligner': item['aligner'],
            'dataset': item['dataset'],
            'abbrev': item['abbrev'],
            'mean_total_height': item['mean_tot'],
            'lower_total_height': item['lower_tot'],
            'upper_total_height': item['upper_tot'],
        }
        
        # Unpack the pandas Series objects into their own columns
        # For example, mean_seg becomes mean_seg_bases_aligned_correct, etc.
        for col_name, value in item['mean_seg'].items():
            base_info[f'mean_seg_{col_name}'] = value
            
        for col_name, value in item['lower_seg'].items():
            base_info[f'lower_seg_{col_name}'] = value
            
        for col_name, value in item['upper_seg'].items():
            base_info[f'upper_seg_{col_name}'] = value
            
        flat_data_for_csv.append(base_info)

    results_df = pd.DataFrame(flat_data_for_csv)

    columns_to_drop = ['mean_total_height', 'lower_total_height', 'upper_total_height']

    df_cleaned = results_df.drop(columns=columns_to_drop)

    df_cleaned.to_csv(output_csv, index=False)

    # ------------------------------------------------------------------ #
    # Step 2: plotting
    # ------------------------------------------------------------------ #
    fig, ax = plt.subplots(figsize=(16, 6))
    width       = 0.12
    bar_gap     = 0.20 * width
    n_datasets  = len(datasets)
    x_aligners  = np.arange(len(aligners))

    for i_ds, dataset in enumerate(datasets):
        offset_ds = (i_ds - n_datasets // 2) * (width + bar_gap)
        if n_datasets % 2 == 0:
            offset_ds += 0.5 * (width + bar_gap)
        xpos = x_aligners + offset_ds

        for j, aligner in enumerate(aligners):
            entry = next(
                (d for d in plot_data
                 if d["aligner"] == aligner and d["dataset"] == dataset),
                None,
            )
            if entry is None:
                continue

            bottom = 0.0
            # ── draw stacked segments ───────────────────────────────────────
            for k, col in enumerate(required_cols):
                seg_height = entry["mean_seg"][col]
                ax.bar(
                    xpos[j],
                    seg_height,
                    width,
                    bottom=bottom,
                    color=colors[k],
                    label=labels[k] if (i_ds == 0 and j == 0) else "",
                )

                if focus == "segment":
                    # CI for this slice alone
                    mean_val = entry["mean_seg"][col]
                    lo = entry["lower_seg"][col]
                    hi = entry["upper_seg"][col]
                    half = (hi - lo) / 2

                    if anchoring == "center":
                        y_center = bottom + seg_height / 2
                    else:  # 'top'
                        y_center = bottom + seg_height

                    err_kw = dict(
                        fmt="none",
                        ecolor="black",
                        capsize=5,
                        elinewidth=1,
                    )
                    if ci_mode == "width":
                        ax.errorbar(
                            xpos[j], y_center, yerr=half, **err_kw,
                        )
                    else:
                        ax.errorbar(
                            xpos[j],
                            y_center,
                            yerr=[[mean_val - lo], [hi - mean_val]],
                            **err_kw,
                        )

                bottom += seg_height

            # ── optional CI for *total* bar height ─────────────────────────
            if focus == "total":
                mean_tot  = entry["mean_tot"]
                lo_tot    = entry["lower_tot"]
                hi_tot    = entry["upper_tot"]
                half_tot  = (hi_tot - lo_tot) / 2
                y_tot     = bottom if anchoring == "top" else bottom / 2

                err_kw = dict(
                    fmt="none",
                    ecolor="black",
                    capsize=5,
                    elinewidth=1,
                )
                if ci_mode == "width":
                    ax.errorbar(
                        xpos[j], y_tot, yerr=half_tot, **err_kw,
                    )
                else:
                    ax.errorbar(
                        xpos[j],
                        y_tot,
                        yerr=[[mean_tot - lo_tot], [hi_tot - mean_tot]],
                        **err_kw,
                    )

    # ── cosmetics ───────────────────────────────────────────────────────────
    ax.set_xticks(x_aligners)
    ax.set_xticklabels(aligners)
    ylabel = (
        "Base counts"
        if method == "counts"
        else "Proportion of total bases"
    )
    ax.set_ylabel(ylabel)

    ci_txt = {
        "width":     "±½ CI width",
        "fullrange": "95 % CI range",
    }[ci_mode]

    if focus == "total":
        ax.set_title(
            f"Baruzzo base-level metrics per aligner "
            f"(Accu{ci_txt}; {n_subsets} bootstraps × {sample_size} reads)"
        )
    else:
        ax.set_title(
            f"Baruzzo base-level metrics per aligner "
            f"({ci_txt}; {n_subsets} bootstraps × {sample_size} reads)"
        )
    # legend & dataset labels underneath
    handles = [plt.Rectangle((0, 0), 1, 1, color=c) for c in colors]
    ax.legend(
        handles, labels, title="Category",
        loc="lower center",
        bbox_to_anchor=(0.5, -0.25),
        ncol=4,
        frameon=True,
    )

    # ── cosmetics ─────────────────────────────────────────────────────────
    ax.set_xticks([])                     # suppress default tick labels

    ylabel = "Base mean counts across replicates" if method == "counts" else "Proportion of total bases mean across replicates"
    ax.set_ylabel(ylabel)

    ci_txt = {"width": "±½ CI width", "fullrange": "95 % CI range"}[ci_mode]
    # ax.set_title(
    #     f"Baruzzo base-level metrics per aligner "
    #     f"({ci_txt}; 500 bootstraps × 20M reads × 100 base)"
    # )

    # --- two-level x-axis labelling --------------------------------------
    y_dataset = -max(ax.get_ylim()) * 0.02   # 6 % below axis
    y_aligner = -max(ax.get_ylim()) * 0.06   # aligner a bit lower

    # dataset tags (one per individual bar)
    for i_ds, ds in enumerate(datasets):
        offset = (i_ds - n_datasets // 2) * (width + bar_gap)
        if n_datasets % 2 == 0:
            offset += 0.5 * (width + bar_gap)

        for j, _ in enumerate(aligners):
            ax.text(
                x_aligners[j] + offset,
                y_dataset,
                abbrev[ds],
                ha="center",
                va="top",
